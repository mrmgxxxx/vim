### Recommended .gitconfig

```
[color]
    branch = auto
    ui = auto
    status = auto
[alias]
    st = status
    co = checkout
    ll = log --graph --abbrev-commit --decorate --format=format:'%C(red)%h%C(reset) - %C(bold yellow)%d%C(reset) %C(bold green)%s%C(reset) %C(blue)- %an%C(reset) %C(blue)%aD%C(reset) %C(blue)(%ar)%C(reset)' --all
[core]
    editor = vim
```
